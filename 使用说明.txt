请在当前桌面上，以管理员身份打开 PowerShell，完整复制并运行以下修复代码：
code
Powershell
Write-Host "=== 正在执行极限精简系统专用记事本底层修复 ===" -ForegroundColor Cyan

# 1. 抓取当前用户 .txt 被死锁绑定的 ProgId (通常是 AppX 开头的一串哈希)
$ucPath = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Explorer\FileExts\.txt\UserChoice"
$currentProgId = $null
if (Test-Path $ucPath) {
    $currentProgId = (Get-ItemProperty -Path $ucPath -ErrorAction SilentlyContinue).ProgId
}

Write-Host "检测到当前锁定的 ProgId 为: $currentProgId" -ForegroundColor Yellow

# 2. 如果存在 ProgId，直接在当前用户与系统级对其进行强制覆盖（拔除 DelegateExecute，重定向到 notepad.exe）
if ($currentProgId) {
    $hkcuCmd = "HKCU:\Software\Classes\$currentProgId\shell\open\command"
    if (-not (Test-Path $hkcuCmd)) {
        New-Item -Path $hkcuCmd -Force | Out-Null
    }
    Set-ItemProperty -Path $hkcuCmd -Name "(Default)" -Value "`"C:\Windows\System32\notepad.exe`" `"%1`""
    Remove-ItemProperty -Path $hkcuCmd -Name "DelegateExecute" -ErrorAction SilentlyContinue

    $hklmCmd = "HKLM:\SOFTWARE\Classes\$currentProgId\shell\open\command"
    if (Test-Path $hklmCmd) {
        Set-ItemProperty -Path $hklmCmd -Name "(Default)" -Value "`"C:\Windows\System32\notepad.exe`" `"%1`""
        Remove-ItemProperty -Path $hklmCmd -Name "DelegateExecute" -ErrorAction SilentlyContinue
    }
    Write-Host "[成功] 已成功接管并斩断针对 $currentProgId 的 UWP 唤醒死锁！" -ForegroundColor Green
}

# 3. 扫描并清洗所有残留的 Notepad 相关 AppX 劫持项
Get-ChildItem -Path "HKLM:\SOFTWARE\Classes" -Filter "AppX*" -ErrorAction SilentlyContinue | ForEach-Object {
    $cmdPath = Join-Path $_.PSPath "shell\open\command"
    if (Test-Path $cmdPath) {
        $cmdVal = (Get-ItemProperty -Path $cmdPath -ErrorAction SilentlyContinue).'(Default)'
        if ($cmdVal -like "*Notepad*" -or $cmdVal -like "*WindowsApps*") {
            Set-ItemProperty -Path $cmdPath -Name "(Default)" -Value "`"C:\Windows\System32\notepad.exe`" `"%1`""
            Remove-ItemProperty -Path $cmdPath -Name "DelegateExecute" -ErrorAction SilentlyContinue
        }
    }
}

# 4. 强制修复基础类 txtfile 与 txtfilelegacy
$notepadCmd = "`"C:\Windows\System32\notepad.exe`" `"%1`""
reg add "HKLM\SOFTWARE\Classes\.txt" /ve /t REG_SZ /d "txtfilelegacy" /f | Out-Null
reg add "HKLM\SOFTWARE\Classes\txtfilelegacy\shell\open\command" /ve /t REG_SZ /d $notepadCmd /f | Out-Null
reg add "HKLM\SOFTWARE\Classes\txtfile\shell\open\command" /ve /t REG_SZ /d $notepadCmd /f | Out-Null

# 5. 尝试通过底层 ACL 剥离强删 UserChoice（双重保险，如果被驱动拦截会自动跳过并依赖上述劫持）
try {
    $extKey = [Microsoft.Win32.Registry]::CurrentUser.OpenSubKey("Software\Microsoft\Windows\CurrentVersion\Explorer\FileExts\.txt", $true)
    if ($extKey) {
        $ucKey = $extKey.OpenSubKey("UserChoice", [Microsoft.Win32.RegistryKeyPermissionCheck]::ReadWriteSubTree, [System.Security.AccessControl.RegistryRights]::ChangePermissions)
        if ($ucKey) {
            $sec = $ucKey.GetAccessControl()
            $rules = $sec.GetAccessRules($true, $true, [System.Security.Principal.NTAccount])
            foreach ($r in $rules) {
                if ($r.AccessControlType -eq [System.Security.AccessControl.AccessControlType]::Deny) {
                    $sec.RemoveAccessRuleSpecific($r)
                }
            }
            $ucKey.SetAccessControl($sec)
            $ucKey.Close()
        }
        $extKey.DeleteSubKeyTree("UserChoice", $false)
        $extKey.Close()
        Write-Host "[成功] UserChoice 保护已解除并彻底抹除！" -ForegroundColor Green
    }
} catch {
    Write-Host "[提示] UserChoice 受驱动强保护无法删除，已通过步骤 2 成功实现命令劫持。" -ForegroundColor Gray
}

# 6. 重启资源管理器
Write-Host "正在重启资源管理器生效..." -ForegroundColor Cyan
Stop-Process -Name explorer -Force
